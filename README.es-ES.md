# Cómo usar:
## crear un entorno virtual 
virtualenv venv 

## activarlo e instalar requisitos
- source venv/bin/activate
- pip install -r requirements.txt 
- python -m spacy download en_core_web_sm

## ejecutar el archivo de la aplicación
python app.py
